//@version=5
strategy("EMA Auto Trader", overlay=true, initial_capital=10000)

fastLength = input.int(20, title="EMA Fast")
slowLength = input.int(50, title="EMA Slow")

fastEMA = ta.ema(close, fastLength)
slowEMA = ta.ema(close, slowLength)

plot(fastEMA, color=color.green, linewidth=2, title="EMA 20")
plot(slowEMA, color=color.red, linewidth=2, title="EMA 50")

buySignal = ta.crossover(fastEMA, slowEMA)
sellSignal = ta.crossunder(fastEMA, slowEMA)

if buySignal
    strategy.close("SELL")
    strategy.entry("BUY", strategy.long)
    alert("BUY", alert.freq_once_per_bar_close)

if sellSignal
    strategy.close("BUY")
    strategy.entry("SELL", strategy.short)
    alert("SELL", alert.freq_once_per_bar_close)

plotshape(buySignal, title="BUY", style=shape.labelup, text="BUY", color=color.green, size=size.small)

plotshape(sellSignal, title="SELL", style=shape.labeldown, text="SELL", color=color.red, size=size.small)
