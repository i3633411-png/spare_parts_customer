//====================================================
// قراءة ملف الإشارة
//====================================================
string ReadSignal()
{
   string signal = "";

   int file = FileOpen("signal.txt", FILE_READ | FILE_TXT);

   if(file != INVALID_HANDLE)
   {
      signal = FileReadString(file);
      FileClose(file);
   }

   return(signal);
}
//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
   SignalText = ReadSignal();

   Comment(
      "AutoTraderEA\n\n",
      "Signal = ", SignalText
   );
}
