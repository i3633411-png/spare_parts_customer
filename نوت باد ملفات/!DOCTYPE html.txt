  let busy = false;

  // Joystick vector, each in -1..1. dx = left/right, dy = up(-1)/down(+1).
  let aimDX = 0, aimDY = -0.35;

  // ===== Joystick drag handling =====
  let dragging = false;
  function setHandle(dx, dy){
    joyHandle.style.left = (50 + dx*50) + '%';
    joyHandle.style.top = (50 + dy*50) + '%';
  }
  function pointerToVector(clientX, clientY){
    const rect = joystick.getBoundingClientRect();
    const cx = rect.left + rect.width/2;
    const cy = rect.top + rect.height/2;
    let dx = (clientX - cx) / (rect.width/2);
    let dy = (clientY - cy) / (rect.height/2);
    const mag = Math.hypot(dx, dy);
    if(mag > 1){ dx /= mag; dy /= mag; }
    return { dx, dy };
  }
  function startDrag(e){
    if(busy || state.attempts >= MAX_ATTEMPTS) return;
    dragging = true;
    joystick.style.cursor = 'grabbing';
    moveDrag(e);
  }
  function moveDrag(e){
    if(!dragging) return;
    const point = e.touches ? e.touches[0] : e;
    const { dx, dy } = pointerToVector(point.clientX, point.clientY);
    aimDX = dx; aimDY = dy;
    setHandle(dx, dy);
    updateAimPreview();
    e.preventDefault && e.preventDefault();
  }
  function endDrag(){
    dragging = false;
    joystick.style.cursor = 'grab';
  }
  joystick.addEventListener('mousedown', startDrag);
  window.addEventListener('mousemove', moveDrag);
  window.addEventListener('mouseup', endDrag);
  joystick.addEventListener('touchstart', startDrag, {passive:false});
  window.addEventListener('touchmove', moveDrag, {passive:false});
  window.addEventListener('touchend', endDrag);

  // Map the joystick vector to a target point inside the goal frame (percentages), full 2D freedom.
  function computeTarget(dx, dy){
    const magnitude = Math.min(1, Math.hypot(dx, dy));
    const x = Math.max(4, Math.min(96, 50 + dx * 48));
    let y = Math.max(3, Math.min(95, 50 + dy * 48));
    let overTheBar = false;
    if(magnitude > 0.93 && y < 14){
      overTheBar = true;
    }
    return { x, y, overTheBar, magnitude };
  }

  function updateAimPreview(){
    const t = computeTarget(aimDX, aimDY);
    powerOut.textContent = Math.round(t.magnitude*100) + '%';

    // Arrow at the ball previews the shot direction/height
    const rotDeg = aimDX * 55;
    const lengthScale = 0.7 + t.magnitude*0.9;
    aimArrow.style.transform = `translate(-50%,0) rotate(${rotDeg}deg) scaleY(${lengthScale})`;
    aimArrow.style.opacity = 0.55 + t.magnitude*0.4;

    if(t.overTheBar){
      aimMarker.style.opacity = '0.25';
      aimMarker.style.top = '-6%';
      aimMarker.style.left = t.x + '%';
    } else {
      aimMarker.style.opacity = '0.9';
      aimMarker.style.top = t.y + '%';
      aimMarker.style.left = t.x + '%';
    }
  }

  function isInTargetCircle(x, y){
    // target circle center roughly at x~80%, y~28% within the goal-frame box, radius ~11 (tight — precision required)
    const cx = 80, cy = 28, r = 11;
    const d = Math.hypot(x-cx, y-cy);
    return d <= r;
  }

  function zoneFor(x, y){
    const top = y < 44;
    if(top){
      if(x < 35) return 'tl';
      if(x < 65) return 'tc';
      return 'tr';
    } else {
      if(x < 35) return 'bl';
      if(x < 65) return 'bc';
      return 'br';
    }
  }
  const zonePoints = { tl:30, tc:20, tr:40, bl:30, bc:10, br:30 };

  // Keeper's guessed horizontal position (0-100), biased toward the target-circle side —
  // and now reacts across a wide span rather than one exact zone, making the keeper hard to beat.
  function keeperGuessX(inTarget){
    // Keeper anticipates the target-circle corner most of the time, otherwise reads center-mass.
    const base = inTarget ? 76 : (42 + Math.random()*16);
    // add some natural variance in the keeper's read
    const variance = (Math.random()-0.5) * 22;
    return Math.max(14, Math.min(86, base + variance));
  }
  // Keeper's guessed vertical position, loosely tracking the shot height so the whole body dives/jumps.
  function keeperGuessY(targetY){
    const variance = (Math.random()-0.5) * 22;
    return Math.max(2, Math.min(70, targetY + variance));
  }

  function fireConfetti(){
    const colors = ['#D6A544','#F7F5EE','#8C6F2E','#EDEAE0'];
    for(let i=0;i<24;i++){
      const p = document.createElement('div');
      p.className = 'confetti-piece';
      p.style.left = (15+Math.random()*70) + '%';
      p.style.background = colors[Math.floor(Math.random()*colors.length)];
      p.style.animationName = 'confettiFall';
      p.style.animationDelay = (Math.random()*0.15) + 's';
      p.style.animationDuration = (0.9+Math.random()*0.6) + 's';
      p.style.borderRadius = Math.random()>0.5 ? '50%' : '2px';
      confettiLayer.appendChild(p);
      setTimeout(()=>p.remove(), 1800);
    }
  }

  function celebrate(bigWin){
    field.classList.remove('celebrating'); void field.offsetWidth; field.classList.add('celebrating');
    netEl.classList.remove('shake'); void netEl.offsetWidth; netEl.classList.add('shake');
    celebrateFlash.classList.remove('on'); void celebrateFlash.offsetWidth; celebrateFlash.classList.add('on');
    celebrateDim.classList.remove('on'); void celebrateDim.offsetWidth; celebrateDim.classList.add('on');
    goalText.textContent = bigWin ? 'هدف من الزاوية المستحيلة!' : 'هـدف!';
    goalText.classList.remove('pop'); void goalText.offsetWidth; goalText.classList.add('pop');
    fireConfetti();
  }

  function showBanner(text, cls){
    resultBanner.textContent = text;
    resultBanner.className = 'result-banner show ' + cls;
    setTimeout(()=> resultBanner.classList.remove('show'), 1500);
  }

  function addLog(icon, desc, pts, cls){
    if(logList.querySelector('.log-empty')) logList.innerHTML = '';
    const row = document.createElement('div');
    row.className = 'log-item';
    row.innerHTML = `<div class="log-icon">${icon}</div><div class="log-desc">${desc}</div><div class="log-pts ${cls}">${pts}</div>`;
    logList.prepend(row);
  }

  function updateHUD(){
    comboVal.textContent = '×' + state.combo;
    scoreVal.textContent = state.score;
    attemptsVal.textContent = Math.max(MAX_ATTEMPTS - state.attempts, 0);
    bestVal.textContent = state.best;
    const done = state.attempts >= MAX_ATTEMPTS;
    shootBtn.disabled = done || busy;
    if(done){
      shootBtn.textContent = 'انتهت المحاولات';
    } else {
      shootBtn.textContent = 'سدّد';
    }
  }

  function shoot(){
    if(busy || state.attempts >= MAX_ATTEMPTS) return;
    busy = true;
    state.attempts++;

    const dx = aimDX, dy = aimDY;
    const t = computeTarget(dx, dy);

    const fRect = field.getBoundingClientRect();
    const goalFrame = document.getElementById('goalFrame');
    const gRect = goalFrame.getBoundingClientRect();

    if(t.overTheBar){
      const bx = gRect.left - fRect.left + gRect.width * (t.x/100);
      ball.classList.add('shooting');
      requestAnimationFrame(()=>{
        ball.style.left = bx + 'px';
        ball.style.bottom = (fRect.height * 0.02) + 'px';
        ball.style.top = '-6%';
        ball.style.width='12px'; ball.style.height='12px';
      });
      setTimeout(()=>{
        state.combo = 1;
        showBanner('فوق العارضة! خفف القوة', 'miss');
        addLog('⬆️', 'فوق العارضة', 'فاتت', 'down');
        finishShot(700);
      }, 420);
      return;
    }

    // Extreme angles clip the post and miss outright — precision matters
    const nearPost = t.x < 12 || t.x > 88;
    if(nearPost && Math.random() < 0.55){
      const bx = gRect.left - fRect.left + gRect.width * (t.x/100);
      const by = gRect.top - fRect.top + gRect.height * (t.y/100);
      ball.classList.add('shooting');
      requestAnimationFrame(()=>{
        ball.style.top = 'auto';
        ball.style.left = bx + 'px';
        ball.style.bottom = (fRect.height - by) + 'px';
        ball.style.width='13px'; ball.style.height='13px';
      });
      setTimeout(()=>{
        state.combo = 1;
        showBanner('ارتطمت بالقائم!', 'miss');
        addLog('🥅', 'ارتطام بالقائم', 'فاتت', 'down');
        finishShot(700);
      }, 420);
      return;
    }

    const inTarget = isInTargetCircle(t.x, t.y);
    const zone = zoneFor(t.x, t.y);
    const basePts = inTarget ? 100 : zonePoints[zone];

    const keeperX = keeperGuessX(inTarget);
    const keeperY = keeperGuessY(t.y); // used for save-probability accuracy and the block point
    keeper.style.left = keeperX + '%';
    keeper.classList.remove('dive-left','dive-right','jump-up');
    if(keeperX < t.x - 3) keeper.classList.add('dive-right');
    if(keeperX > t.x + 3) keeper.classList.add('dive-left');
    if(t.y < 35) keeper.classList.add('jump-up');

    // Decide the outcome BEFORE animating, so the ball's path itself confirms it —
    // a save visibly stops short and is parried away; a goal travels all the way into the net.
    const ddx = Math.abs(t.x - keeperX);
    const ddy = Math.abs(t.y - keeperY);
    const distance = Math.hypot(ddx, ddy * 0.7);
    let saveChance = Math.max(0.16, 0.88 - distance * 0.009);
    if(inTarget) saveChance = Math.min(0.95, saveChance + 0.22);
    const saved = Math.random() < saveChance;

    const bx = gRect.left - fRect.left + gRect.width * (t.x/100);
    const by = gRect.top - fRect.top + gRect.height * (t.y/100);

    if(saved){
      // Ball travels only to the keeper's interception point, not all the way to the target
      const blockX = keeperX + (t.x - keeperX) * 0.55;
      const blockY = t.y + (keeperY - t.y) * 0.35;
      const kbx = gRect.left - fRect.left + gRect.width * (blockX/100);
      const kby = gRect.top - fRect.top + gRect.height * (blockY/100);

      ball.classList.add('shooting');
      requestAnimationFrame(()=>{
        ball.style.top = 'auto';
        ball.style.left = kbx + 'px';
        ball.style.bottom = (fRect.height - kby) + 'px';
        ball.style.width='13px'; ball.style.height='13px';
      });

      setTimeout(()=>{
        // Parry: ball bounces away from the keeper, down and to a side — unmistakably not a goal
        const deflectX = Math.max(8, Math.min(92, blockX + (Math.random()>0.5?1:-1) * (14 + Math.random()*10)));
        const deflectY = Math.min(96, blockY + 22 + Math.random()*10);
        const dbx = gRect.left - fRect.left + gRect.width * (deflectX/100);
        const dby = gRect.top - fRect.top + gRect.height * (deflectY/100);
        ball.style.transition = 'all 0.3s cubic-bezier(.3,.6,.4,1)';
        ball.style.left = dbx + 'px';
        ball.style.bottom = (fRect.height - dby) + 'px';
        ball.style.width='10px'; ball.style.height='10px';

        state.combo = 1;
        showBanner('تصدّى لها الحارس!', 'saved');
        addLog('🧤', inTarget ? 'تسديدة نحو الزاوية — تصدٍّ' : `تسديدة ${zone.toUpperCase()} — تصدٍّ`, '+0', 'down');
        finishShot(950);
      }, 320);
      return;
    }

    // Goal: ball travels all the way to the target and settles inside the net
    ball.classList.add('shooting');
    requestAnimationFrame(()=>{
      ball.style.top = 'auto';
      ball.style.left = bx + 'px';
      ball.style.bottom = (fRect.height - by) + 'px';
      ball.style.width='13px'; ball.style.height='13px';
    });

    setTimeout(()=>{
      const pts = basePts * state.combo;
      state.score += pts;
      state.best = Math.max(state.best, state.score);
      celebrate(inTarget);
      showBanner(inTarget ? `هدف من الزاوية المستحيلة! +${pts}` : `هدف! +${pts}`, 'goal');
      addLog(inTarget ? '🏆' : '⚽', inTarget ? `هدف من الزاوية المستحيلة — كومبو ×${state.combo}` : `هدف (${zone.toUpperCase()}) — كومبو ×${state.combo}`, '+'+pts, 'up');
      if(state.combo < MAX_COMBO) state.combo++;
      // Ball stays settled in the net during the celebration hold
      finishShot(1700);
    }, 480);
  }

  function finishShot(delay){
    updateHUD();
    setTimeout(()=>{
      ball.style.transition = '';
      ball.classList.remove('shooting');
      ball.style.left = '50%'; ball.style.bottom = '24%'; ball.style.top = 'auto';
      ball.style.width='20px'; ball.style.height='20px';
      keeper.style.left = '50%';
      keeper.classList.remove('dive-left','dive-right','jump-up');
      busy = false;
      updateHUD();
    }, delay || 700);
  }

  shootBtn.addEventListener('click', shoot);
  resetBtn.addEventListener('click', ()=>{
    state = { score:0, attempts:0, combo:1, best:state.best };
    aimDX = 0; aimDY = -0.35;
    setHandle(aimDX, aimDY);
    updateAimPreview();
    updateHUD();
    logList.innerHTML = '<div class="log-empty">لم تبدأ بعد</div>';
    showBanner('تمت إعادة التحدي', 'goal');
  });

  setHandle(aimDX, aimDY);
  updateAimPreview();
  updateHUD();
})();
</script>
