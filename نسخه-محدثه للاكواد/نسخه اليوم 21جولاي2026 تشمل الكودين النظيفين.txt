//+------------------------------------------------------------------+
#property strict
#include <Trade/Trade.mqh>

CTrade trade;

string last_signal = "";

//+------------------------------------------------------------------+
void OnTick()
{
   string signal = ReadSignal();

   if(signal == "" || signal == last_signal)
      return;

   Print("🔥 New Signal: ", signal);

   string parts[];
   int count = StringSplit(signal, ',', parts);

   if(count != 2)
   {
      Print("❌ Invalid format");
      return;
   }

   string symbol = parts[0];
   string action = parts[1];

   if(StringFind(_Symbol, symbol) != -1)
   {
      if(action == "BUY")
      {
         CloseSell();

         if(WaitClose())
         {
            OpenBuy();
         }
      }
      else if(action == "SELL")
      {
         CloseBuy();

         if(WaitClose())
         {
            OpenSell();
         }
      }

      last_signal = signal;
   }
}
//+------------------------------------------------------------------+

string ReadSignal()
{
   int handle = FileOpen("Signal.txt", FILE_READ | FILE_TXT | FILE_ANSI);

   if(handle == INVALID_HANDLE)
   {
      Print("❌ FILE NOT FOUND");
      return "";
   }

   string text = FileReadString(handle);
   FileClose(handle);

   StringTrimLeft(text);
   StringTrimRight(text);

   return text;
}
//+------------------------------------------------------------------+

// ⏳ انتظار إغلاق الصفقة فعليًا
bool WaitClose()
{
   for(int i=0; i<10; i++)
   {
      Sleep(500);

      if(!PositionSelect(_Symbol))
         return true;
   }

   Print("❌ Failed to confirm close");
   return false;
}
//+------------------------------------------------------------------+

void OpenBuy()
{
   Print("🟢 OPEN BUY");
   trade.Buy(0.10, _Symbol);
}
//+------------------------------------------------------------------+

void OpenSell()
{
   Print("🔴 OPEN SELL");
   trade.Sell(0.10, _Symbol);
}
//+------------------------------------------------------------------+

void CloseBuy()
{
   if(PositionSelect(_Symbol))
   {
      if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY)
      {
         Print("❌ CLOSE BUY");
         trade.PositionClose(_Symbol);
      }
   }
}
//+------------------------------------------------------------------+

void CloseSell()
{
   if(PositionSelect(_Symbol))
   {
      if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_SELL)
      {
         Print("❌ CLOSE SELL");
         trade.PositionClose(_Symbol);
      }
   }
}

IMAP_SERVER = "imap.gmail.com"
CHECK_INTERVAL = 5

# ================= STATE =================
last_signal = ""
last_email_id = ""

# ================= استخراج الإشارة =================
def extract_signal(text):
    text = text.upper()

    if "BUY" in text or "LONG" in text:
        return "BUY"
    elif "SELL" in text or "SHORT" in text:
        return "SELL"

    return ""

# ================= الاتصال =================
mail = imaplib.IMAP4_SSL(IMAP_SERVER)
mail.login(EMAIL, PASSWORD)

print("🚀 Python Bot Started...")

# ================= LOOP =================
while True:
    try:
        mail.select("inbox")

        # 🔥 فقط الإيميلات الجديدة
        result, data = mail.search(None, "UNSEEN")
        ids = data[0].split()

        if ids:
            latest_id = ids[-1]

            # منع تكرار نفس الإيميل
            if latest_id == last_email_id:
                time.sleep(CHECK_INTERVAL)
                continue

            last_email_id = latest_id

            result, msg_data = mail.fetch(latest_id, "(RFC822)")
            raw_email = msg_data[0][1]

            msg = email.message_from_bytes(raw_email)

            subject = msg["subject"] if msg["subject"] else ""
            body = ""

            if msg.is_multipart():
                for part in msg.walk():
                    if part.get_content_type() == "text/plain":
                        body = part.get_payload(decode=True).decode(errors="ignore")
                        break
            else:
                body = msg.get_payload(decode=True).decode(errors="ignore")

            full_text = subject + " " + body

            signal = extract_signal(full_text)

            if signal != "" and signal != last_signal:

                with open(file_path, "w") as f:
                    symbol = "BTCUSD"
                    f.write(symbol + "," + signal)

                print("🔥 NEW SIGNAL:", signal)

                last_signal = signal
            else:
                print("⏳ No new valid signal")

        else:
            print("⏳ Waiting...")

        time.sleep(CHECK_INTERVAL)

    except Exception as e:
        print("❌ ERROR:", e)
        time.sleep(CHECK_INTERVAL)
