//+------------------------------------------------------------------+
#property strict
#include <Trade/Trade.mqh>

CTrade trade;
datetime last_execution_time = 0;

int OnInit()
{
   // تحديد نسبة الانزلاق المسموح بها لضمان قبول Exness لأوامر الإغلاق والفتح
   trade.SetDeviationInPoints(50);
   return(INIT_SUCCEEDED);
}

void OnTick()
{
   // 1. قراءة الملف
   int handle = FileOpen("Signal.txt", FILE_READ | FILE_TXT | FILE_ANSI);
   if(handle == INVALID_HANDLE) return;

   string signal = FileReadString(handle);
   FileClose(handle);

   StringTrimLeft(signal);
   StringTrimRight(signal);

   if(signal == "") return;

   // حماية زمنيّة: منع تكرار المعالجة خلال ثانيتين
   if(TimeCurrent() - last_execution_time < 2)
   {
      ClearSignalFile();
      return;
   }

   Print("🔔 RECEIVED NEW SIGNAL: ", signal);

   string parts[];
   if(StringSplit(signal, ',', parts) != 2)
   {
      ClearSignalFile();
      return;
   }

   string action = parts[1];
   StringToUpper(action);

   // 2. تفريغ الملف فوراً لمنع القراءة المكررة
   ClearSignalFile();
   last_execution_time = TimeCurrent();

   // 3. إغلاق جراحي لكافة الصفقات المفتوحة
   CloseAllPositionsAbsolute();

   // 🛑 4. شرط أمان صارم: عدم الفتح إلا إذا تأكدنا أن الحساب أصبح 0 صفقات
   if(PositionsTotal() == 0)
   {
      if(action == "BUY")
      {
         if(trade.Buy(0.03, _Symbol))
            Print("✅ SUCCESS: Opened BUY");
         else
            Print("❌ FAIL BUY. Error: ", trade.ResultRetcodeDescription());
      }
      else if(action == "SELL")
      {
         if(trade.Sell(0.03, _Symbol))
            Print("✅ SUCCESS: Opened SELL");
         else
            Print("❌ FAIL SELL. Error: ", trade.ResultRetcodeDescription());
      }
      else if(action == "CLOSE")
      {
         Print("🧹 Account Cleared via CLOSE Signal");
      }
   }
   else
   {
      Print("⚠️ WARNING: Could not open new position because old positions failed to close. Total open: ", PositionsTotal());
   }
}

// دالة مسح محتوى الملف
void ClearSignalFile()
{
   int handle = FileOpen("Signal.txt", FILE_WRITE | FILE_TXT | FILE_ANSI);
   if(handle != INVALID_HANDLE)
   {
      FileWriteString(handle, "");
      FileClose(handle);
   }
}

// دالة إغلاق كافة الصفقات مع المتابعة
void CloseAllPositionsAbsolute()
{
   int attempts = 0;
   while(PositionsTotal() > 0 && attempts < 10)
   {
      for(int i = PositionsTotal() - 1; i >= 0; i--)
      {
         ulong ticket = PositionGetTicket(i);
         if(ticket > 0)
         {
            if(!trade.PositionClose(ticket))
            {
               Print("❌ Close failed for ticket ", ticket, " - Error: ", trade.ResultRetcodeDescription());
            }
         }
      }
      Sleep(300); // مهلة زمنية لاستجابة السيرفر وتحديث القائمة
      attempts++;
   }
}
